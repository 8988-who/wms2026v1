# WMS 前端新增模块模板（通用）

> 基于 `wmsui` 项目现有代码规范，新增任意业务模块时按此模板创建文件。
>
> **占位符说明：**
> - `{业务模块}` — 业务大类目录名，如 `warehouse`、`rcs`、`system`
> - `{模块名}` — 具体模块名（kebab-case），如 `wms-location`、`rcs-task`
> - `{组件名}` — Vue 组件名（PascalCase），如 `WmsLocation`、`RcsTask`
> - `{API对象}` — API 对象名（PascalCase），如 `WmsLocationAPI`、`RcsTaskAPI`

## 一、目录结构

```
src/
├── api/
│   └── {业务模块}/              # 所属业务大类
│       └── {模块名}/            # 新模块（kebab-case）
│           ├── index.ts        # API 定义
│           └── types.ts        # 类型定义
└── views/
    └── {业务模块}/
        └── {模块名}/
            └── index.vue       # 页面视图
```

## 二、API 类型定义 — `src/api/{业务模块}/{模块名}/types.ts`

```typescript
/**
 * {组件名} 类型定义
 */

import type { BaseQueryParams } from "@/api/common";

/** 查询参数 */
export interface {组件名}QueryParams extends BaseQueryParams {
  keywords?: string;
  status?: number;
}

/** 表单对象（新增/编辑） */
export interface {组件名}Form {
  id?: number;
  name?: string;
  status?: number;
  remark?: string;
}

/** 列表项（表格展示） */
export interface {组件名}Item {
  id?: number;
  name?: string;
  status?: number;
  remark?: string;
  createdByName?: string;
  createdTime?: string;
  updatedByName?: string;
  updatedTime?: string;
}

/** 详情对象 */
export type {组件名}Detail = {组件名}Item & {组件名}Form;
```

## 三、API 方法定义 — `src/api/{业务模块}/{模块名}/index.ts`

```typescript
import request from "@/utils/request";
import type { {组件名}Form, {组件名}QueryParams, {组件名}Item } from "./types";
import type { PageResult } from "@/api/common";

const BASE_URL = "/api/v1/{模块名}";

const {API对象} = {
  /** 获取分页数据 */
  getPage(queryParams?: {组件名}QueryParams) {
    return request<unknown, PageResult<{组件名}Item>>({
      url: `${BASE_URL}`,
      method: "get",
      params: queryParams,
    });
  },
  /** 获取表单数据 */
  getFormData(id: string) {
    return request<unknown, {组件名}Form>({
      url: `${BASE_URL}/${id}/form`,
      method: "get",
    });
  },
  /** 新增 */
  create(data: {组件名}Form) {
    return request({
      url: `${BASE_URL}`,
      method: "post",
      data,
    });
  },
  /** 更新 */
  update(id: string, data: {组件名}Form) {
    return request({
      url: `${BASE_URL}/${id}`,
      method: "put",
      data,
    });
  },
  /** 批量删除，多个以英文逗号(,)分割 */
  deleteByIds(ids: string) {
    return request({
      url: `${BASE_URL}/${ids}`,
      method: "delete",
    });
  },
  /** 批量更新状态（启用/停用） */
  updateStatus(data: { ids: number[]; status: number }) {
    return request({
      url: `${BASE_URL}/status`,
      method: "put",
      data,
    });
  },
};

export default {API对象};
export * from "./types";
```

## 四、页面视图 — `src/views/{业务模块}/{模块名}/index.vue`

```vue
<template>
  <div class="page-container">
    <!-- 搜索 -->
    <el-card class="page-search" shadow="never">
      <el-form ref="queryFormRef" :model="params" :inline="true">
        <el-form-item label="关键字" prop="keywords">
          <el-input
            v-model="params.keywords"
            placeholder="名称"
            clearable
            @keyup.enter="handleQuery"
          />
        </el-form-item>
        <el-form-item label="状态" prop="status">
          <el-select v-model="params.status" placeholder="状态" clearable>
            <el-option label="启用" :value="1" />
            <el-option label="禁用" :value="0" />
          </el-select>
        </el-form-item>
        <el-form-item>
          <el-button type="primary" @click="handleQuery">搜索</el-button>
          <el-button @click="handleResetQuery">重置</el-button>
        </el-form-item>
      </el-form>
    </el-card>

    <!-- 表格 -->
    <el-card class="page-content" shadow="never">
      <div class="page-toolbar">
        <div class="page-toolbar__left">
          <el-button
            v-hasPerm="['{业务模块}:{模块名}:create']"
            type="primary"
            @click="handleOpenDialog()"
          >
            新增
          </el-button>
          <el-dropdown
            v-hasPerm="['{业务模块}:{模块名}:update', '{业务模块}:{模块名}:delete']"
            :disabled="!hasSelection"
            @command="handleBatchCommand"
            style="margin-left: 10px"
          >
            <el-button :disabled="!hasSelection">
              批量操作<el-icon class="el-icon--right"><ArrowDown /></el-icon>
            </el-button>
            <template #dropdown>
              <el-dropdown-menu>
                <el-dropdown-item command="enable" v-hasPerm="['{业务模块}:{模块名}:update']">
                  启用
                </el-dropdown-item>
                <el-dropdown-item command="disable" v-hasPerm="['{业务模块}:{模块名}:update']">
                  停用
                </el-dropdown-item>
                <el-dropdown-item
                  command="delete"
                  v-hasPerm="['{业务模块}:{模块名}:delete']"
                  divided
                >
                  删除
                </el-dropdown-item>
              </el-dropdown-menu>
            </template>
          </el-dropdown>
        </div>
        <div class="page-toolbar__right">
          <el-tooltip content="刷新" placement="top">
            <el-button class="page-icon-btn" @click="fetchData">
              <el-icon><Refresh /></el-icon>
            </el-button>
          </el-tooltip>
        </div>
      </div>

      <div class="page-table-wrapper">
        <el-table
          v-loading="loading"
          :data="tableData"
          @selection-change="handleSelectionChange"
        >
          <el-table-column type="selection" width="50" align="center" />
          <el-table-column label="编号" prop="id" width="80" />
          <el-table-column label="名称" prop="name" />
          <el-table-column label="状态" prop="status" width="100">
            <template #default="{ row }">
              <el-tag :type="row.status === 1 ? 'success' : 'info'">
                {{ row.status === 1 ? "启用" : "禁用" }}
              </el-tag>
            </template>
          </el-table-column>
          <el-table-column label="创建人" prop="createdByName" width="120" />
          <el-table-column label="创建时间" prop="createdTime" width="180" />
          <el-table-column label="操作" width="180" fixed="right">
            <template #default="{ row }">
              <el-button
                v-hasPerm="['{业务模块}:{模块名}:update']"
                type="primary"
                link
                @click="handleOpenDialog(row.id)"
              >
                编辑
              </el-button>
              <el-button
                v-hasPerm="['{业务模块}:{模块名}:delete']"
                type="danger"
                link
                @click="handleDelete(row.id)"
              >
                删除
              </el-button>
            </template>
          </el-table-column>
        </el-table>
      </div>

      <pagination
        v-model:total="total"
        v-model:page="params.pageNum"
        v-model:limit="params.pageSize"
        @pagination="fetchData"
      />
    </el-card>

    <!-- 弹窗 -->
    <el-dialog
      v-model="dialog.visible"
      :title="dialog.title"
      width="500px"
      @close="handleCloseDialog"
    >
      <el-form ref="formRef" :model="formData" :rules="rules" label-width="80px">
        <el-form-item label="名称" prop="name">
          <el-input v-model="formData.name" placeholder="请输入名称" />
        </el-form-item>
        <el-form-item label="状态" prop="status">
          <el-radio-group v-model="formData.status">
            <el-radio :value="1">启用</el-radio>
            <el-radio :value="0">禁用</el-radio>
          </el-radio-group>
        </el-form-item>
        <el-form-item label="备注" prop="remark">
          <el-input v-model="formData.remark" type="textarea" />
        </el-form-item>
      </el-form>
      <template #footer>
        <el-button @click="handleCloseDialog">取消</el-button>
        <el-button type="primary" :loading="submitting" @click="handleSubmit">
          确定
        </el-button>
      </template>
    </el-dialog>
  </div>
</template>

<script setup lang="ts">
import {API对象} from "@/api/{业务模块}/{模块名}";
import type {
  {组件名}QueryParams,
  {组件名}Form,
  {组件名}Item,
} from "@/api/{业务模块}/{模块名}";
import { useTableSelection } from "@/composables/useTableSelection";

defineOptions({ name: "{组件名}" });

// 查询参数
const params = reactive<{组件名}QueryParams>({
  pageNum: 1,
  pageSize: 10,
  keywords: undefined,
  status: undefined,
});

// 表格状态
const loading = ref(false);
const tableData = ref<{组件名}Item[]>([]);
const total = ref(0);

// 表格选择
const { selectedIds, hasSelection, handleSelectionChange } =
  useTableSelection<{组件名}Item>();

// 弹窗状态
const dialog = reactive({
  visible: false,
  title: "",
});
const formRef = ref();
const submitting = ref(false);
const formData = reactive<{组件名}Form>({});
const rules = {
  name: [{ required: true, message: "请输入名称", trigger: "blur" }],
};

// 查询
function handleQuery() {
  params.pageNum = 1;
  fetchData();
}

// 重置
function handleResetQuery() {
  Object.assign(params, {
    keywords: undefined,
    status: undefined,
  });
  handleQuery();
}

// 获取数据
async function fetchData() {
  loading.value = true;
  try {
    const res = await {API对象}.getPage(params);
    tableData.value = res.list;
    total.value = res.total;
  } finally {
    loading.value = false;
  }
}

// 打开弹窗
async function handleOpenDialog(id?: number) {
  if (id) {
    dialog.title = "编辑";
    const data = await {API对象}.getFormData(String(id));
    Object.assign(formData, data);
  } else {
    dialog.title = "新增";
    Object.assign(formData, { status: 1 });
  }
  dialog.visible = true;
}

// 关闭弹窗
function handleCloseDialog() {
  dialog.visible = false;
  formRef.value?.resetFields();
  Object.keys(formData).forEach((key) => delete (formData as any)[key]);
}

// 提交
async function handleSubmit() {
  await formRef.value?.validate();
  submitting.value = true;
  try {
    if (formData.id) {
      await {API对象}.update(String(formData.id), formData);
    } else {
      await {API对象}.create(formData);
    }
    ElMessage.success("保存成功");
    handleCloseDialog();
    fetchData();
  } finally {
    submitting.value = false;
  }
}

// 删除
async function handleDelete(id: number) {
  await ElMessageBox.confirm("确认删除该记录？", "警告", {
    type: "warning",
  });
  await {API对象}.deleteByIds(String(id));
  ElMessage.success("删除成功");
  fetchData();
}

// 批量操作
async function handleBatchCommand(command: string) {
  const ids = selectedIds.value;
  if (command === "delete") {
    await ElMessageBox.confirm("确认删除选中的记录？", "警告", {
      type: "warning",
    });
    await {API对象}.deleteByIds(ids.join(","));
  } else {
    await {API对象}.updateStatus({
      ids: ids as number[],
      status: command === "enable" ? 1 : 0,
    });
  }
  ElMessage.success("操作成功");
  fetchData();
}

onMounted(fetchData);
</script>
```

## 七、关键规则速查

| 规则 | 说明 |
|---|---|
| Vue API | **不写 import**，由 unplugin-auto-import 自动导入 |
| Element Plus 组件 | **不写 import**，自动导入 |
| 路径别名 | `@/` 指向 `src/` |
| 权限标识 | `{业务模块}:{模块名}:{动作}`（如 `warehouse:wms-location:create`） |
| 组件 name | `defineOptions({ name: "{组件名}" })`，需与路由 name 一致 |
| 批量操作 | 统一用 `el-dropdown` 下拉菜单，不用独立按钮 |
| 分页组件 | `<pagination>` 全局注册，用 `v-model` 绑定 |
| 表格选择 | 使用 `useTableSelection` 组合式函数 |
| 目录命名 | kebab-case（如 `wms-location`、`rcs-task`） |
| 组件命名 | PascalCase（如 `WmsLocation`、`RcsTask`） |
| API 导出 | `export default {API对象}` + `export * from "./types"` |
| 类型集中 | 每个模块的 `types.ts` 集中定义所有接口类型 |
| 通用类型 | `ApiResult`、`PageResult`、`BaseQueryParams` 从 `@/api/common` 导入 |
